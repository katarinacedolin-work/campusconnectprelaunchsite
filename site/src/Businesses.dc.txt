    const reviews = stores.reduce((n, s) => n + s.reviews, 0);
    const followers = stores.reduce((n, s) => n + s.followers, 0);
    const avg = stores.reduce((n, s) => n + parseFloat(s.rating) * s.reviews, 0) / reviews;
    const statDefs = [
      { value: String(stores.length), label: 'Student stores' },
      { value: fmt(orders), label: 'Orders completed' },
      { value: fmt(reviews), label: 'Customer reviews' },
      { value: avg.toFixed(1), label: 'Average rating' },
      { value: fmt(followers), label: 'Store followers' }
    ];
    const stats = statDefs.map((s, i) => Object.assign({}, s, { divider: i === 0 ? '0' : '1px solid rgba(250,250,247,0.24)', padLeft: i === 0 ? '0' : '28px' }));

    const top = ranked[0].orders;
    const accents = ['#F6C945', '#B28CFF', '#67C7FF'];
    const leaders = ranked.map((s, i) => {
      const d = details[s.id] || { byline: '', photo: '', photoAlt: '', initials: s.name.slice(0, 2).toUpperCase() };
      return {
        rank: '0' + (i + 1), name: s.name, byline: d.byline, photo: d.photo, noPhoto: !d.photo, photoAlt: d.photoAlt, initials: d.initials,
        orders: fmt(s.orders), rating: s.rating, reviews: fmt(s.reviews), followers: fmt(s.followers),

        bg: i === 0 ? 'rgba(246,201,69,0.08)' : '#16171C', border: i === 0 ? 'rgba(246,201,69,0.35)' : 'rgba(250,250,247,0.10)'
      };
    });
