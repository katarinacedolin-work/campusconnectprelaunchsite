<script type="text/x-dc" data-dc-script data-props='{"$preview":{"width":1440,"height":4700}}'>
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { slot: 1, demoOpen: false, phoneOpen: false };
  }
  setTilt(e, maxX, maxY) {
    const el = e.currentTarget;
    if (!el || !el.getBoundingClientRect) return;
    const r = el.getBoundingClientRect();
    const x = (e.clientX - r.left) / r.width;
    const y = (e.clientY - r.top) / r.height;
    el.style.setProperty('--ry', ((x - 0.5) * maxX).toFixed(2) + 'deg');
    el.style.setProperty('--rx', ((0.5 - y) * maxY).toFixed(2) + 'deg');
    el.style.setProperty('--mx', (x * 100).toFixed(1) + '%');
    el.style.setProperty('--my', (y * 100).toFixed(1) + '%');
  }
  resetTilt(e, mx, my) {
    const el = e.currentTarget;
    if (!el || !el.style) return;
    el.style.setProperty('--rx', '0deg');
    el.style.setProperty('--ry', '0deg');
    el.style.setProperty('--mx', mx);
    el.style.setProperty('--my', my);
  }
  renderVals() {
    const st = this.state || {};
    const iconLayers = [];
    for (let i = 0; i < 14; i++) {
      const t = i / 13;
      iconLayers.push({ z: (i * 2.4 - 16).toFixed(1), bg: 'rgb(' + Math.round(14 + t * 26) + ',' + Math.round(30 + t * 40) + ',' + Math.round(130 + t * 105) + ')' });
    }
    const logoLayers = [];
    for (let i = 0; i < 12; i++) {
      const t = i / 11;
      logoLayers.push({ z: (i * 2.2 - 12).toFixed(1), color: 'rgb(' + Math.round(28 + t * 162) + ',' + Math.round(52 + t * 152) + ',' + Math.round(170 + t * 85) + ')' });
    }
    const words = ['Founder curriculum', 'Expert office hours', 'Investor introductions', 'Student marketplace', 'Trust & community', 'Your campus is full of talent'];
    const marquee = words.concat(words).map((w) => ({ text: w }));
    const slotIdx = st.slot == null ? 1 : st.slot;
    const slots = ['Tue · 3:00 PM', 'Wed · 11:30 AM', 'Thu · 4:15 PM'].map((label, i) => {
      const on = i === slotIdx;
      return { label, selected: on, tag: on ? 'Booked ✓' : 'Open', bg: on ? '#3157FF' : 'rgba(250,250,247,0.04)', border: on ? '#3157FF' : 'rgba(250,250,247,0.18)', pick: () => this.setState({ slot: i }) };
    });
    return {
      iconLayers, logoLayers, marquee, slots,
      demoOpen: !!st.demoOpen,
      phoneOpen: !!st.phoneOpen,
      openPhone: () => this.setState({ phoneOpen: true }),
      closePhone: () => this.setState({ phoneOpen: false }),
      backdropPhone: (e) => { if (e && e.target === e.currentTarget) this.setState({ phoneOpen: false }); },
      openDemo: () => this.setState({ demoOpen: true }),
      closeDemo: () => this.setState({ demoOpen: false }),
      backdrop: (e) => { if (e && e.target === e.currentTarget) this.setState({ demoOpen: false }); },
      heroMove: (e) => this.setTilt(e, 16, 10),
      heroLeave: (e) => this.resetTilt(e, '72%', '40%'),
      tilt: (e) => this.setTilt(e, 7, 6),
      untilt: (e) => this.resetTilt(e, '50%', '50%')
    };
  }
}
</script>
