  setTilt(e, maxX, maxY, hero) {
    const src = e.currentTarget;
    if (!src || !src.getBoundingClientRect) return;
    const tgt = hero && src.closest ? (src.closest('.cc-hero') || src) : src;
    const r = src.getBoundingClientRect();
    const x = (e.clientX - r.left) / r.width;
    const y = (e.clientY - r.top) / r.height;
    const R = tgt.getBoundingClientRect();
    tgt.style.setProperty('--ry', ((x - 0.5) * maxX).toFixed(2) + 'deg');
    tgt.style.setProperty('--rx', ((0.5 - y) * maxY).toFixed(2) + 'deg');
    tgt.style.setProperty('--mx', (((e.clientX - R.left) / R.width) * 100).toFixed(1) + '%');
    tgt.style.setProperty('--my', (((e.clientY - R.top) / R.height) * 100).toFixed(1) + '%');
  }
  resetTilt(e, mx, my, hero) {
    const src = e.currentTarget;
    if (!src || !src.style) return;
    const tgt = hero && src.closest ? (src.closest('.cc-hero') || src) : src;
    tgt.style.setProperty('--rx', '0deg');
    tgt.style.setProperty('--ry', '0deg');
    tgt.style.setProperty('--mx', mx);
    tgt.style.setProperty('--my', my);
  }
