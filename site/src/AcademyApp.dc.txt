      const p = pctOf(c);

    const nextMod = cur ? cur.mods.findIndex((_, i) => !modDone(cur, i)) : 0;
    const course = cur ? {
      title: cur.name, desc: cur.blurb, kicker: cur.code + (cur.id === 'crash' ? ' · Start here' : ''), hours: cur.hours.replace(' hours', ''), modulesN: cur.mods.length, doneN: curDone, pct: curPct, cap: cur.cap,
      action: curPct === 0 ? 'Start course' : complete(cur) ? 'Review course' : 'Continue course',
      continue: nextMod === -1 ? this.go('cert', { courseId: cur.id }) : openMod(cur.id, nextMod < 0 ? 0 : nextMod),
      certificate: this.go('cert', { courseId: cur.id }),
      certLabel: complete(cur) ? 'View certificate' : 'Certificate',
      capMark: st.cap[cur.id] ? '✓' : '★', capCls: st.cap[cur.id] ? 'ac-tick' : 'ac-num',
      capNote: st.cap[cur.id] ? 'Submitted' : cur.mods.length + ' modules come first',
      submitCap: () => {
        if (this.state.cap[cur.id]) { this.flash('Capstone already submitted'); return; }
        if (doneN(cur) < cur.mods.length) { this.flash('Finish the modules before the capstone'); return; }
        const cap = Object.assign({}, this.state.cap); cap[cur.id] = true; this.setState({ cap: cap }); this.flash('Capstone submitted');
      },
      back: openCourse(cur.id),
      mods: cur.mods.map((m, i) => ({ n: i + 1, title: m.t, num: modDone(cur, i) ? '✓' : String(i + 1), numCls: modDone(cur, i) ? 'ac-tick' : 'ac-num', meta: m.cards.length + ' topics' + (modDone(cur, i) ? ' · Completed' : ''), open: openMod(cur.id, i) }))
    } : { mods: [], title: '', desc: '', kicker: '', hours: '', modulesN: 0, doneN: 0, pct: 0, cap: '', action: '', continue: this.go('courses'), certificate: this.go('courses'), certLabel: '', capMark: '', capCls: '', capNote: '', submitCap: () => {}, back: this.go('courses') };

    const mod = cur && cur.mods[st.mod] ? cur.mods[st.mod] : (cur ? cur.mods[0] : { t: '', h: '', cards: [] });
    const modIndex = cur && cur.mods[st.mod] ? st.mod : 0;
    const lesson = {
      kicker: (cur ? cur.code : '') + ' · Module ' + (modIndex + 1) + ' of ' + (cur ? cur.mods.length : 0),
      title: mod.t, hook: mod.h, note: 'The ideas in this module, before the activity and the summary in the full course.',
      topics: (mod.cards || []).map((t) => ({ t: t })),
      markLabel: cur && modDone(cur, modIndex) ? 'Module complete ✓' : 'Mark module complete',
      mark: () => {
        if (!cur || modDone(cur, modIndex)) return;
        const done = Object.assign({}, this.state.done); done[cur.id + '-' + modIndex] = true;
        this.setState({ done: done, lastId: cur.id });
        this.flash('Module marked complete');
      },
      nextLabel: cur && modIndex + 1 < cur.mods.length ? 'Next module' : 'Back to syllabus',
      next: cur && modIndex + 1 < cur.mods.length ? openMod(cur.id, modIndex + 1) : openCourse(cur ? cur.id : 'crash')
    };
    const leftN = cur ? (cur.mods.length - curDone) + (st.cap[cur.id] ? 0 : 1) : 0;
    const cert = { ok: !!(cur && complete(cur)), locked: !(cur && complete(cur)), line: cur ? cur.code + ' ' + cur.name : '', meta: cur ? cur.mods.length + ' modules and a capstone · ' + cur.hours : '', left: leftN + (leftN === 1 ? ' requirement still open.' : ' requirements still open.') };

    const last = byId(st.lastId) || coursesRaw[0];
    const lastPct = last ? pctOf(last) : 0;
    const resume = last ? Object.assign({
      kicker: lastPct ? 'Continue learning' : 'Start here',
      title: last.name, code: last.code, hours: last.hours, where: last.mods.length + ' modules',
      badge: last.code.replace(' ', '\n'),
      pct: lastPct, status: lastPct <= 0 ? 'Not started' : complete(last) ? 'Complete' : lastPct + '% complete',
      action: complete(last) ? 'Review →' : lastPct ? 'Resume →' : 'Start →',
      open: lastPct ? openCourse(last.id) : openMod(last.id, 0)
    }, bits(last.stage)) : { kicker: '', title: '', code: '', hours: '', where: '', badge: '', pct: 0, status: '', action: '', open: this.go('courses'), stage: '', stageLabel: '', on1: '', on2: '', on3: '', on4: '' };

    const docFilters = filters([['all', 'All documents'], ['foundations', 'Foundations'], ['marketing', 'Marketing'], ['finance', 'Finance'], ['operations', 'Operations'], ['setup', 'Business setup']], st.docFilter, 'docFilter');
    const docs = DOCS.filter((d) => (st.docFilter === 'all' || d.slug === st.docFilter) && match(d.title + ' ' + d.desc)).map((d) => Object.assign({ title: d.title, desc: d.desc, topic: d.topic, meta: d.meta, cta: d.kind === 'Checklist' ? 'Open checklist' : 'Read guide', open: (e) => { this.top(e); this.setState({ view: 'doc', docKind: 'guide', docId: d.id }); } }, bits(d.stage)));
    const shelfFilters = filters([['all', 'All volumes'], ['foundations', 'Foundations'], ['marketing', 'Marketing'], ['finance', 'Finance'], ['operations', 'Operations'], ['setup', 'Business setup']], st.shelfFilter, 'shelfFilter');
    const initials = (name) => name.split(/[^A-Za-z]+/).filter(Boolean).map((w) => w[0]).slice(0, 2).join('');
    const shelf = SHELF.map((d, i) => ({
      vol: 'Volume ' + String(i + 1).padStart(2, '0'), title: d.title, desc: d.desc, author: d.author, role: d.role, stat: d.stat || 'Campus Connect Exclusive', exclusive: !!d.exclusive, topic: d.topic, text: d.title + ' ' + d.desc + ' ' + d.author,
      hasPhoto: !!d.photo, noPhoto: !d.photo, photo: d.photo, initials: initials(d.author),
      open: (e) => { this.top(e); this.setState({ view: 'doc', docKind: 'volume', docId: d.id }); }
    })).filter((d) => (st.shelfFilter === 'all' || d.topic === st.shelfFilter) && match(d.text));
    const stageFilters = filters([['all', 'All levels'], ['idea', 'Idea'], ['launch', 'Launch'], ['growing', 'Growing'], ['scaling', 'Scaling']], st.tplStage, 'tplStage');
    const tplFilters = filters([['all', 'All categories']].concat(TPL_GROUPS), st.tplGroup, 'tplGroup');
    const tplItems = TEMPLATE_DATA.filter((t) => (st.tplGroup === 'all' || t.group === st.tplGroup) && (st.tplStage === 'all' || (t.essential || []).indexOf(st.tplStage) !== -1) && match(t.code + ' ' + t.title + ' ' + t.about));
    const tplGroups = TPL_GROUPS.map((g) => ({
      name: g[1],
      items: tplItems.filter((t) => t.group === g[0]).map((t) => Object.assign({
        code: t.code, title: t.title, desc: t.about, kind: t.format === 'Excel' ? 'Spreadsheet' : 'Document',
        open: (e) => { this.top(e); this.setState({ view: 'doc', docKind: 'template', docId: t.id }); }

    let doc = { kicker: '', title: '', note: '', sections: [], blanks: [], takeaway: '', hasTakeaway: false, legal: false, hasAuthor: false, hasPhoto: false, noPhoto: true, photo: '', initials: '', author: '', role: '', saveLabel: '', save: () => {}, back: this.go('documents'), backName: 'Documents', stage: '', stageLabel: '', on1: '', on2: '', on3: '', on4: '' };
    if (st.docKind === 'guide') {
      const item = DOCS.find((d) => d.id === st.docId) || DOCS[0];
      doc = Object.assign(doc, bits(item.stage), { kicker: item.kicker, title: item.title, note: item.note, takeaway: item.takeaway, hasTakeaway: true, sections: item.sections.map((m) => ({ h: m[0], p: m[1] })), blanks: (item.blanks || []).map((m) => ({ h: m[0], p: m[1] })), saveLabel: st.saved[item.id] ? 'Saved to library ✓' : 'Save to library', save: () => { const s = Object.assign({}, this.state.saved); s[item.id] = true; this.setState({ saved: s }); this.flash('Saved to your library'); }, back: this.go('documents'), backName: 'Documents' });
    } else if (st.docKind === 'volume') {
      const item = SHELF.find((d) => d.id === st.docId) || SHELF[0];
      const n = String(SHELF.indexOf(item) + 1).padStart(2, '0');
      doc = Object.assign(doc, { kicker: (item.exclusive ? 'Campus Connect Exclusive · ' : '') + 'The Founders Shelf · Volume ' + n, title: item.title, note: item.desc, takeaway: item.takeaway, hasTakeaway: true, hasAuthor: true, hasPhoto: !!item.photo, noPhoto: !item.photo, photo: item.photo, initials: initials(item.author), author: item.author, role: item.role, sections: item.sections.map((m) => ({ h: m[0], p: m[1] })), saveLabel: st.saved[item.id] ? 'Saved to library ✓' : 'Save to library', save: () => { const s = Object.assign({}, this.state.saved); s[item.id] = true; this.setState({ saved: s }); this.flash('Saved to your library'); }, back: this.go('shelf'), backName: 'Founders Shelf' });
    } else if (st.docKind === 'template') {
      const item = TEMPLATE_DATA.find((t) => t.id === st.docId) || TEMPLATE_DATA[0] || { id: '', code: '', title: '', about: '', format: 'Word', group: '', stage: 'idea' };
      const group = (TPL_GROUPS.find((g) => g[0] === item.group) || ['', ''])[1];
      const word = (stage) => ({ idea: 'Essential', launch: 'Essential', growing: 'Useful', scaling: 'Useful' }[stage] || 'Skip for now');
      doc = Object.assign(doc, bits(item.stage), { kicker: 'Template · ' + group, title: item.title, note: item.about, legal: item.group === 'legal', sections: [{ h: item.code + ' · ' + (item.format === 'Excel' ? 'Excel workbook' : 'Word document'), p: 'In the full Business Hub this downloads as a file you fill in and keep with your business records.' }], blanks: [{ h: 'This stage', p: word(item.stage) + ' while you are at ' + (STAGE_L[item.stage] || 'this stage') + '.' }], saveLabel: 'Download template', save: () => this.flash('Template download opens in the full Business Hub'), back: this.go('templates'), backName: 'Templates' });

      { id: 'jordan', initials: 'JP', name: 'Jordan Patel', area: 'Brand & Marketing', when: 'Tomorrow · 11:30 AM', tag: 'marketing', stage: 'launch' },
      { id: 'riley', initials: 'RS', name: 'Riley Santos', area: 'Finance & Funding', when: 'Fri · 9:00 AM', tag: 'finance', stage: 'scaling' }

'' : 'background:#1c3148', cls: 'ac-cta ac-locked' + (done ? ' done' : ''), label: label, act: () => {} }, bits(p.stage));

    }));
    const bookedCount = Object.keys(st.booked).length;

    return {
      screenH: (this.props && this.props.screenH) || '720px',
      screenRadius: (this.props && this.props.screenRadius) || '0',
      side: side,
      v: { home: view === 'home', courses: view === 'courses', course: view === 'course', lesson: view === 'lesson', cert: view === 'cert', documents: view === 'documents', shelf: view === 'shelf', templates: view === 'templates', doc: view === 'doc', office: view === 'office', investors: view === 'investors', founder: view === 'founder', about: view === 'about', profile: view === 'profile' },
      goHome: this.go('home'), goCourses: this.go('courses'), goOffice: this.go('office'), goAbout: this.go('about'), goProfile: this.go('profile'), goInvestors: this.go('investors'), goFounder: this.go('founder'), goTemplates: this.go('templates'),
      locked: () => {},
      openMarketing: openCourse('mkt'),
      noNotifs: () => this.flash('No new notifications'),
      editProfile: () => this.flash('Profile editor opened'),
      showBookings: () => { const n = Object.keys(this.state.booked).length; if (!n) { this.flash('No sessions booked yet'); return; } this.setState({ bookingsOnly: true, expertFilter: 'all' }); this.flash(n === 1 ? '1 session booked' : n + ' sessions booked'); },
      searchQ: st.searchQ, onSearch: (e) => this.setState({ searchQ: e.target.value }),
      inProgress: coursesRaw.filter((c) => pctOf(c) > 0 && !complete(c)).length,
      docsReady: 12 + Object.keys(st.saved).length,
      sessions: Math.max(1, bookedCount),
      overall: overall, remaining: 100 - overall,
      readyNote: overall === 0 ? 'You have not started a course yet.' : overall === 100 ? 'Every Startup Foundations course is complete.' : 'You are ' + overall + '% through Startup Foundations.',
      resume: resume,
      courseFilters: courseFilters, courses: courses, coursesEmpty: courses.length === 0,
      course: course, lesson: lesson, cert: cert,
      docFilters: docFilters, docs: docs, docsEmpty: docs.length === 0,
      shelfFilters: shelfFilters, shelf: shelf, shelfEmpty: shelf.length === 0,
      stageFilters: stageFilters, tplFilters: tplFilters, tplGroups: tplGroups, tplEmpty: tplGroups.length === 0,
      doc: doc,
      expertFilters: expertFilters, experts: experts, expertsEmpty: experts.length === 0,
      people: people,
      founderHeadline: st.founderSaved && String(founder.offer || '').trim() ? founder.offer.trim() : 'Make it easy for the right person to say “tell me more.”',
      founderPct: Math.round(filled / 7 * 100), founderFilled: filled, founderFields: founderFields,
      saveFounder: (e) => { if (e && e.preventDefault) e.preventDefault(); },
      bookTick: bookedCount ? '✓' : '!', bookTickCls: 'ac-tick' + (bookedCount ? '' : ' pending'),
      healthBook: bookedCount ? 'Expert hour booked' : 'Book your next expert hour',
      toast: st.toast
    };
  }
