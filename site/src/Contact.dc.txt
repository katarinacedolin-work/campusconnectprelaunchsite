<script type="text/x-dc" data-dc-script data-props='{"$preview":{"width":1440,"height":3000}}'>
class Component extends DCLogic {
  constructor(props) {
    super(props);
    this.state = { role: 'uni', sent: false, faq: 0 };
  }
  setTilt(e, maxX, maxY) {
    const el = e.currentTarget;
    if (!el || !el.getBoundingClientRect) return;
    const r = el.getBoundingClientRect();
    const x = (e.clientX - r.left) / r.width;
    const y = (e.clientY - r.top) / r.height;
    el.style.setProperty('--ry', ((x - 0.5) * maxX).toFixed(2) + 'deg');
    el.style.setProperty('--rx', ((0.5 - y) * maxY).toFixed(2) + 'deg');
    el.style.setProperty('--mx', (x * 100).toFixed(1) + '%');
    el.style.setProperty('--my', (y * 100).toFixed(1) + '%');
  }
  resetTilt(e, mx, my) {
    const el = e.currentTarget;
    if (!el || !el.style) return;
    el.style.setProperty('--rx', '0deg');
    el.style.setProperty('--ry', '0deg');
    el.style.setProperty('--mx', mx);
    el.style.setProperty('--my', my);
  }
  renderVals() {
    const st = this.state || {};
    const role = st.role || 'uni';
    const roleDefs = [
      { id: 'uni', label: 'University', org: 'University or department', ph: 'e.g. College of Business' },
      { id: 'inv', label: 'Investor', org: 'Firm or fund', ph: 'e.g. Seed fund or angel group' },
      { id: 'men', label: 'Mentor', org: 'Company or area of expertise', ph: 'e.g. Marketing, operations, law' },
      { id: 'stu', label: 'Student founder', org: 'Your business', ph: 'What are you building?' }
    ];
    const roles = roleDefs.map((r) => {
      const on = r.id === role;
      return { label: r.label, selected: on, bg: on ? '#151515' : 'transparent', fg: on ? '#FAFAF7' : '#151515', border: on ? '#151515' : '#DEDED7', pick: () => this.setState({ role: r.id }) };
    });
    const cur = roleDefs.find((r) => r.id === role) || roleDefs[0];

    const logoLayers = [];
    for (let i = 0; i < 14; i++) {
      const t = i / 13;
      logoLayers.push({ z: (i * 2.2 - 14).toFixed(1), color: 'rgb(' + Math.round(14 + t * 35) + ',' + Math.round(24 + t * 63) + ',' + Math.round(90 + t * 165) + ')' });
    }

    const faqDefs = [
      { q: 'How does a university partnership work?', a: 'We start with an intro call, set the program up for your campus — curriculum, mentors and your school services — then launch with your students.' },
      { q: 'When does Campus Connect launch?', a: 'The marketplace launches in January 2027 at San Diego State, starting with 50 founding student sellers.' },
      { q: 'Who can buy and sell on the marketplace?', a: 'Verified students. Universities can also list school-provided services, and local businesses can share deals for students.' },
      { q: 'How can investors get involved?', a: 'Browse profiles of student founders and their businesses, accept one-on-one calls with the founders you want to meet, or hold office hours as a mentor.' }
    ];
    const openIdx = st.faq == null ? -1 : st.faq;
    const faqs = faqDefs.map((f, i) => {
      const on = i === openIdx;
      return { q: f.q, a: f.a, open: on, rot: on ? 45 : 0, iconBg: on ? '#3157FF' : '#EFEFEA', iconFg: on ? '#FAFAF7' : '#151515', toggle: () => this.setState({ faq: on ? -1 : i }) };
    });

    return {
      roles, orgLabel: cur.org, orgPlaceholder: cur.ph, logoLayers, faqs,
      sent: !!st.sent, notSent: !st.sent,
      submit: (e) => { if (e && e.preventDefault) e.preventDefault(); this.setState({ sent: true }); },
      reset: () => this.setState({ sent: false }),
      heroMove: (e) => this.setTilt(e, 24, 14),
      heroLeave: (e) => this.resetTilt(e, '72%', '50%')
    };
  }
}
